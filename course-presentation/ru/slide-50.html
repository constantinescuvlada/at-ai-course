attest.py — интеграционные тесты

Тест — это код, который автоматически проверяет другой код. assert означает «громко упади, если это не так» — поэтому успешный прогон доказывает, что все утверждения выполнились. Интеграционные тесты проверяют целые сценарии (API вместе с базой данных) и ловят проблемы «стыковки», которые маленький «юнит»-тест бы пропустил.

Каждый тест обращается к работающему API через requests, а затем заглядывает прямо в базу данных, чтобы убедиться, что изменение действительно произошло. Вот подготовительная часть файла и один полный тест:

attest.pyFILE
import requests
import psycopg2

BASE_URL = "http://localhost:8080"
CONN_INFO = "host=localhost port=5433 dbname=at_data_db user=vlada password=abc"

def getDbCon():
    return psycopg2.connect(CONN_INFO)

def fetch_by_id_from_db(page_id):
    with getDbCon() as conn:
        with conn.cursor() as cursor:
            cursor.execute("SELECT page, title, context FROM at_data WHERE id = %s", (page_id,))
            return cursor.fetchone()

def test_add_page():
    print("[INFO] TEST 2: Testing ADD POST /at_data/ ...")
    payload = {
        "page": "Test Page",
        "title": "Test Title",
        "context": "Test Context",
        "page_vector": [0.1]*10,
        "title_vector": [0.2]*10,
        "context_vector": [0.3]*10
    }

    response = requests.post(f"{BASE_URL}/at_data/", json=payload)
    assert response.status_code == 201
    new_id = response.json()["id"]

    db_row = fetch_by_id_from_db(new_id)
    assert db_row == ("Test Page", "Test Title", "Test Context")

    with getDbCon() as conn:
        with conn.cursor() as cursor:
            print("[CLEANUP] deleting test info from database...")
            cursor.execute("DELETE FROM at_data WHERE id = %s", (new_id,))
            conn.commit()
    print("[PASS] Test 2 passed")

with ... as ...: открывает ресурс (соединение, курсор) и автоматически закрывает его в конце блока — более аккуратный родственник try/finally. В настоящем файле есть ещё пять таких же тестов: список, отклонение невалидных данных, обновление, удаление, получение по id.

Хорошие тесты убирают за собой — этот в конце удаляет свою тестовую строку, поэтому прогон набора никогда не замусоривает ваши данные.
