Ответ с памятью диалога

HTTP — протокол без состояния: между запросами сервер забывает всё. Поэтому для настоящего диалога клиент должен каждый раз отправлять всю историю чата. Этот POST-вариант принимает её как список строк.

atassistantapi.py — продолжениеFILE
@app.post("/at_data/answer")
def generate_ollama_answer(messages: List[str] = Body(...)):
    try:
        history = messages[:-1]
        question = messages[-1]
        print(f"history: {history}") #DEBUG
        print(f"question: {question}") #DEBUG
        query_vector = calcEmbed(question)

        conn = getDbCon()
        cur = conn.cursor()

        cur.execute("""
            SELECT context, context_vector <=> %s::vector AS distance
            FROM at_data
            WHERE context_vector <=> %s::vector < 0.5
            ORDER BY context_vector <=> %s::vector
            LIMIT 5
        """, (query_vector, query_vector, query_vector))

        rows = cur.fetchall()
        if not rows:
            print(f"[ERROR] No similar context found for question {question}")
            raise HTTPException(status_code=500, detail="No similar context found")

        combined_context = "\n".join([f"Chunk: {row[0]}" for row in rows])

        history_messages = [{"role": "user" if i % 2 == 0 else "assistant", "content": msg}
                            for i, msg in enumerate(history)]
        history_messages.append({
            "role": "user",
            "content": question
        })

        response = ollama.chat(
            model=OLLAMA_MODEL,
            messages=[
                {
                    "role": "system",
                    "content": (
                        "You are a helpful and formal assistant. "
                        "Use the following context to answer the user's question. "
                        "Be concise and professional.\n\n"
                        f"Use only following context as factual information:\n{combined_context}"
                    )
                }
            ] + history_messages
        )

        return {
            "question": question,
            "answer": response["message"]["content"].strip()
        }

    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        cur.close()
        conn.close()

Трюк с чётными/нечётными: сообщения чередуются: пользователь → ассистент → пользователь…, поэтому чётная позиция в списке означает «это сказал пользователь», а нечётная — «это ответил бот». Модель получает полную размеченную историю плюс свежий контекст для последнего вопроса.

Строки #DEBUG выводят каждый запрос в терминал сервера — удобно, пока учитесь; позже удалите их, если станут мешать.
