Эндпоинт ответа: извлечение → генерация

Передаём найденные фрагменты чат-модели вместе со строгим системным промптом и возвращаем её ответ. ollama.chat отправляет список сообщений, каждое со своей ролью: system (правила) и user (вопрос).

atassistantapi.py — продолжениеFILE
@app.get("/at_data/answer")
def generate_ollama_answer(question: str = Query(..., description="User's question")):
    try:
        query_vector = calcEmbed(question)

        conn = getDbCon()
        cur = conn.cursor()

        cur.execute("""
            SELECT context, context_vector <=> %s::vector AS distance
            FROM at_data
            WHERE context_vector <=> %s::vector < 0.5
            ORDER BY context_vector <=> %s::vector
            LIMIT 5
        """, (query_vector, query_vector, query_vector))

        rows = cur.fetchall()
        if not rows:
            raise HTTPException(status_code=404, detail="No similar context found")

        combined_context = "\n".join([f"Chunk: {row[0]}" for row in rows])

        response = ollama.chat(
            model=OLLAMA_MODEL,
            messages=[
                {
                    "role": "system",
                    "content": (
                        "You are a helpful and formal assistant. "
                        "Use the following context to answer the user's question. "
                        "Be concise and professional.\n\n"
                        f"Use only following context as factual informartion:\n{combined_context}"
                    )
                },
                {
                    "role": "user",
                    "content": question
                }
            ]
        )

        return {
            "question": question,
            "answer": response["message"]["content"].strip()
        }

    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        cur.close()
        conn.close()

Эта единственная функция и есть RAG: найденный контекст + вопрос → обоснованный ответ. Если ничего похожего не нашлось, она возвращает 404, вместо того чтобы позволить модели гадать. Дополнительный except HTTPException: raise пробрасывает наши собственные аккуратные ошибки дальше, чтобы они не заворачивались в 500.
