Răspuns cu memoria conversației

HTTP este stateless: serverul uită totul între cereri. Așa că, pentru o conversație reală, clientul trebuie să trimită de fiecare dată întregul istoric al chatului. Această variantă POST îl acceptă ca listă de string-uri.

@app.post("/at_data/answer")
def generate_ollama_answer(messages: List[str] = Body(...)):
    try:
        history = messages[:-1]
        question = messages[-1]
        print(f"history: {history}") #DEBUG
        print(f"question: {question}") #DEBUG
        query_vector = calcEmbed(question)

        conn = getDbCon()
        cur = conn.cursor()

        cur.execute("""
            SELECT context, context_vector <=> %s::vector AS distance
            FROM at_data
            WHERE context_vector <=> %s::vector < 0.5
            ORDER BY context_vector <=> %s::vector
            LIMIT 5
        """, (query_vector, query_vector, query_vector))

        rows = cur.fetchall()
        if not rows:
            print(f"[ERROR] No similar context found for question {question}")
            raise HTTPException(status_code=500, detail="No similar context found")

        combined_context = "\n".join([f"Chunk: {row[0]}" for row in rows])

        history_messages = [{"role": "user" if i % 2 == 0 else "assistant", "content": msg}
                            for i, msg in enumerate(history)]
        history_messages.append({
            "role": "user",
            "content": question
        })

        response = ollama.chat(
            model=OLLAMA_MODEL,
            messages=[
                {
                    "role": "system",
                    "content": (
                        "You are a helpful and formal assistant. "
                        "Use the following context to answer the user's question. "
                        "Be concise and professional.\n\n"
                        f"Use only following context as factual information:\n{combined_context}"
                    )
                }
            ] + history_messages
        )

        return {
            "question": question,
            "answer": response["message"]["content"].strip()
        }

    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    finally:
        cur.close()
        conn.close()

Trucul par/impar: mesajele alternează user → assistant → user…, deci o poziție pară în listă înseamnă „asta a spus utilizatorul", iar una impară — „asta a răspuns botul". Modelul primește întregul istoric etichetat plus context proaspăt pentru cea mai nouă întrebare.

Print-urile #DEBUG afișează fiecare cerere în terminalul serverului — utile cât timp înveți; șterge-le mai târziu dacă devin zgomotoase.
